### Обновленная экономическая логика

```python
class SolowModel:
    def __init__(self):
        self.alpha = 0.3  # доля капитала в производстве
        self.depreciation_rate = 0.05  # норма амортизации
        self.technology_growth = 0.02  # рост технологий
    
    def production_function(self, capital, labor, technology):
        """Производственная функция Y = A * K^α * L^(1-α)"""
        return technology * (capital ** self.alpha) * (labor ** (1 - self.alpha))
    
    def capital_dynamics(self, capital, output, savings_rate):
        """Динамика капитала dK/dt = s*Y - δ*K"""
        investment = savings_rate * output
        depreciation = self.depreciation_rate * capital
        return investment - depreciation
```
